function ema(vals,p=20){let k=2/(p+1),e=vals[0],out=[];vals.forEach((v,i)=>{e=i? v*k+e*(1-k):v;out.push(e)});return out}
function atr(cs=candles){if(cs.length<20)return 0;let a=[];for(let i=1;i<cs.length;i++)a.push(Math.max(cs[i].high-cs[i].low,Math.abs(cs[i].high-cs[i-1].close),Math.abs(cs[i].low-cs[i-1].close)));return a.slice(-14).reduce((x,y)=>x+y,0)/Math.min(14,a.length)}
function sr(cs=candles){let h=[],l=[];for(let i=2;i<cs.length-2;i++){if(cs[i].high>cs[i-1].high&&cs[i].high>cs[i+1].high)h.push(cs[i].high);if(cs[i].low<cs[i-1].low&&cs[i].low<cs[i+1].low)l.push(cs[i].low)}let p=cs.at(-1)?.close||0;let resistance=h.filter(x=>x>p).sort((a,b)=>a-b)[0]||Math.max(...cs.slice(-40).map(x=>x.high));let support=l.filter(x=>x<p).sort((a,b)=>b-a)[0]||Math.min(...cs.slice(-40).map(x=>x.low));return{support,resistance}}
function profitOpportunity(a, quote={}){
 const dir=a.bias==='LONG'?1:a.bias==='SHORT'?-1:0;
 if(!dir)return {edge:-Infinity,opportunity:-Infinity,signal:false,entryReady:false};
 const cs=Array.isArray(a?._cs)&&a._cs.length>=6?a._cs:candles;
 const c=a.c, p1=cs.at(-2), p3=cs.at(-4), p5=cs.at(-6);
 if(!p1||!p3||!p5)return {edge:-Infinity,opportunity:-Infinity,signal:false,entryReady:false};
 const A=Math.max(a.A,Math.abs(c.close)*0.0005,1e-12);
 const slope3=dir*(c.close-p3.close)/A;
 const slope5=dir*(c.close-p5.close)/A;
 const bodyImpulse=dir*(c.close-c.open)/A;
 const emaAlign=dir*(c.close-a.e)/A;
 const emaSlope=dir*a.emaSlope;
 const elapsed=Math.max(0.05,Math.min(1,(Date.now()/1000-c.time)/900));
 const expectedVol=Math.max(0.05,elapsed);
 const volRate=Number(a.vol||0)/expectedVol;
 const volBoost=Math.min(2,Math.max(0,volRate-0.55));
 const breakout=(dir===1&&a.breakoutLong)||(dir===-1&&a.breakoutShort);
 const rejection=(dir===1&&a.rejectLong)||(dir===-1&&a.rejectShort);
 const immediate=dir*(c.close-p1.close)/A;
 const liveImpulse=dir*(c.close-c.open)/A;
 const extension=Math.max(0,emaAlign);
 const candleAligned=(dir===1?c.close>c.open:c.close<c.open);
 const prevAligned=(dir===1?p1.close>p1.open:p1.close<p1.open);
 const trendAligned=emaAlign>-0.25;
 const momentumOk=dir*a.momentum>0.04 || liveImpulse>0.08 || immediate>0.06;
 // Early-entry mode: a forming 15M candle can qualify before its close.
 // Normalize volume by candle age so the engine can detect a fresh momentum ignition.
 const earlyMomentum=liveImpulse>0.08 && immediate>0.03 && volRate>=0.45;
 const continuation=immediate>0.05 && (candleAligned||prevAligned) && volRate>=0.4;
 const bodyOk=Number(a.bodyPct||0)>=0.10 || earlyMomentum || breakout || rejection;
 const volumeOk=Number(a.vol||0)>=0.30 || volRate>=0.45 || earlyMomentum;
 const structureOk=breakout||rejection||earlyMomentum||continuation||(slope3>0.08&&bodyImpulse>0.03)||(trendAligned&&immediate>0.015);
 const notChasing=extension<=2.0 && !(liveImpulse>1.4 && extension>1.15);
 const rangeSetup=a.regime!=='EXTREME VOLATILITY' && (a.regime!=='RANGE'||breakout||rejection||earlyMomentum||continuation);
 const regimeOk=rangeSetup;
 const candleTiming=candleAligned||breakout||rejection||earlyMomentum||immediate>0.015;
 // EARLY ENTRY GATE: prefer the first 45% of a 15M candle. After 65%, only a fresh breakout/impulse can qualify.
 const earlyWindow=elapsed<=0.45,lateCandle=elapsed>0.65;
 const freshBreakout=breakout&&(immediate>0.06||liveImpulse>0.10)&&elapsed<=0.70;
 const timingGate=(earlyWindow&&(earlyMomentum||continuation||candleAligned||breakout||rejection))||freshBreakout||(!lateCandle&&candleTiming);
 const twoCandleConfirm=candleAligned&&prevAligned;
 const ignition=(elapsed<=0.35)&&((candleAligned&&immediate>0.018&&emaAlign>-0.35)||(prevAligned&&immediate>0.045&&volRate>=0.30)||(breakout&&immediate>0.035));
 const reversalRisk=(dir===1&&c.close<a.e&&candleAligned)||(dir===-1&&c.close>a.e&&candleAligned);
 const signal=trendAligned&&momentumOk&&bodyOk&&volumeOk&&structureOk&&notChasing&&regimeOk&&candleTiming&&timingGate&&!reversalRisk&&(twoCandleConfirm||earlyMomentum||ignition||breakout||rejection);
 const edge=slope3*0.24+slope5*0.10+bodyImpulse*0.22+emaAlign*0.10+emaSlope*0.07+volBoost*0.28+immediate*0.24+(breakout?0.8:0)+(rejection?0.45:0)+(earlyMomentum?0.65:0)+(continuation?0.35:0)-(extension>1.15?(extension-1.15)*1.0:0);
 const opportunity=edge*22+Math.min(12,Math.max(0,immediate*7))+Math.min(10,Math.max(0,liveImpulse*7))+Math.min(8,Math.max(0,slope3*3))+(breakout?9:0)+(rejection?4:0)+(earlyMomentum?8:0)+(continuation?4:0)+Math.min(6,Math.max(0,(volRate-0.7)*4));
 const freshness=Math.max(0,1-elapsed);
 const profitEdge=edge+(breakout?.25:0)+(rejection?.15:0)+(earlyMomentum?.25:0)+(ignition?.22:0)+(trendAligned?.08:0);
 const opportunityBoost=(freshness*5)+(ignition?4:0);
 const profitSignal=signal&&elapsed<=0.58&&profitEdge>0.06&&opportunity+opportunityBoost>1.0;
 return {edge,profitEdge,opportunity:opportunity+opportunityBoost,signal:profitSignal,entryReady:profitSignal,slope3,slope5,immediate,liveImpulse,extension,trendAligned,momentumOk,bodyOk,volumeOk,volRate,structureOk,notChasing,regimeOk,earlyMomentum,continuation,ignition,elapsed,freshness};
}

 const momentum=(c.close-p.close)/(A||Math.max(c.close*.001,1));

 const nearRes=Math.abs(c.close-resistance)/(A||1)<1.25,nearSup=Math.abs(c.close-support)/(A||1)<1.25;

 let long=0,short=0;
 long+=trend===1?20:0; short+=trend===-1?20:0;
 long+=emaSlope>0?10:0; short+=emaSlope<0?10:0;
 long+=momentum>0?Math.min(15,Math.max(0,momentum*7)):0; short+=momentum<0?Math.min(15,Math.max(0,-momentum*7)):0;

 if(regime==='RANGE'){long-=8;short-=8} if(regime==='EXTREME VOLATILITY'){long-=12;short-=12}
 long=Math.max(0,Math.min(100,long)); short=Math.max(0,Math.min(100,short));
 const score=Math.max(long,short), bias=long>=short+5?'LONG':short>=long+5?'SHORT':'NEUTRAL';

 if(Math.abs(momentum)<.20)reasons.push('momentum sangat lemah');
 if(vol<.50)reasons.push('volume relatif rendah');
 if(bias==='NEUTRAL')reasons.push('arah belum terbaca');
 if(bias==='LONG'&&c.close<e)reasons.push('di bawah EMA20');
